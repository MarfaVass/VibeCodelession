51async def _build_and_send(target: types.Message, kind: str, task_service,
52 settings_repo, statistics_service, timezone_name: str,
53 exports_dir: Path = EXPORTS_DIR) -> None:
54 """Формирует файл выгрузки (csv|xlsx) и отправляет его документом.
55 
56 Общая точка для /list_csv, /export_excel и кнопок меню экспорта —
57 бизнес-логика формирования файлов не дублируется.
58 Пустая база не блокирует выгрузку: оба формата позволяют создать
59 отчёт без задач (CSV — строка заголовков, Excel — три листа).
60 """
61 tasks = await task_service.tasks.all_for_export()
62 if kind == "csv":
63 path = write_csv_file(tasks, exports_dir, timezone_name)
64 else:
65 stats = await statistics_service.collect()
66 dictionaries = await settings_repo.all_for_export()
67 path = write_xlsx(tasks, stats, dictionaries, exports_dir, timezone_name)
68 
69 caption = CAPTIONS[kind].format(count=len(tasks))
70 try:
71 await target.answer_document(types.FSInputFile(path), caption=clamp_text(caption, 900))
72 logger.info("Файл экспорта отправлен: %s (%d задач)", path.name, len(tasks))
73 except Exception:
74 # Ошибки доставки (сеть/Telegram API) не должны оставлять у
75 # пользователя ощущение успеха — показываем понятное сообщение,
76 # подробности в логах. Файл всё равно удаляется в finally.
77 logger.exception("Не удалось отправить файл экспорта: %s", path.name)
78 try:
79 await target.answer(
80 "⚠️ Не удалось отправить файл выгрузки. "
81 "Попробуйте ещё раз — если не поможет, обратитесь к администратору."
82 )
83 except Exception: # noqa: S110 — чат мог стать недоступным; не критично
84 pass
85 finally:
86 if path.exists():
87 path.unlink(missing_ok=True)
88 logger.info("Временный файл экспорта удалён: %s", path.name)

20async def leave_scenario(state: FSMContext) -> bool:
21 """Завершает активный FSM-сценарий при переходе пользователя в другой раздел.
22 
23 Управляемая очистка (в отличие от state.clear()):
24 1) читаем данные до очистки;
25 2) стираем активное состояние и временные данные незавершённого
26 сценария (черновик задачи, purpose редактирования/админ-ввода);
27 3) восстанавливаем ТОЛЬКО навигационные данные: last_list (контекст
28 исходного списка) и search_text (чтобы продолжал работать поиск
29 и его пагинация после возврата в меню).
30 
31 Возвращает True, если сценарий был активен (для уведомления
32 «Незавершённый сценарий отменён»).
33 """
34 data = await state.get_data()
35 was_active = (await state.get_state() is not None) or bool(data)
36 nav = {
37 key: value for key in ("last_list", "search_text")
38 if (value := data.get(key))
39 }
40 await state.clear()
41 if nav:
42 await state.set_data(nav)
43 return was_active

46async def nav_context(state: FSMContext | None) -> str | None:
47 """last_list из FSM-данных, если он корректен ('kind' или 'kind:value').
48 
49 Некорректный/отсутствующий контекст → None (безопасный fallback
50 в общий список).
51 """
52 if state is None:
53 return None
54 raw = (await state.get_data()).get("last_list")
55 if isinstance(raw, str) and raw.count(":") in (1, 2):
56 return raw
57 return None
